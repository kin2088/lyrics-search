// index.js — 使用 DeepL API 的翻译微服务 const express = require('express'); const app = express(); // DeepL API 密钥（已填入） const DEEPL_API_KEY = '51a8015d-6fea-45e6-8bf0-7336ba4bf26a:fx'; app.use((req, res, next) => { res.setHeader('Access-Control-Allow-Origin', '*'); res.setHeader('Access-Control-Allow-Methods', 'POST, OPTIONS'); res.setHeader('Access-Control-Allow-Headers', 'Content-Type'); if (req.method === 'OPTIONS') return res.status(200).end(); next(); }); app.use(express.json({ limit: '10mb' })); app.post('/translate', async (req, res) => { const { text } = req.body; if (!text) return res.status(400).json({ error: 'Missing text' }); try { const response = await fetch('https://api-free.deepl.com/v2/translate', { method: 'POST', headers: { 'Content-Type': 'application/json', 'Authorization': `DeepL-Auth-Key ${DEEPL_API_KEY}` }, body: JSON.stringify({ text: [text], target_lang: 'ZH' }) }); if (!response.ok) { const errText = await response.text(); console.error('DeepL API 错误:', errText); return res.status(500).json({ error: 'DeepL API 错误: ' + response.status }); } const data = await response.json(); const translated = data.translations?.[0]?.text; if (translated) { return res.json({ translatedText: translated }); } else { return res.status(500).json({ error: '翻译结果为空' }); } } catch (e) { console.error('翻译错误:', e); return res.status(500).json({ error: '翻译失败: ' + e.message }); } }); app.get('/', (req, res) => res.send('翻译服务运行中 (DeepL)')); module.exports = app;
